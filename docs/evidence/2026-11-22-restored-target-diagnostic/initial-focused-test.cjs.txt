function dom(items) {
  const $ = cheerio.load('<main><div class="message-list"></div></main>');
  for (const item of items) {
    const row = $('<article class="message-row"><div class="message-body"></div></article>');
    row.attr('data-message-id', item.id);
    if (item.self) row.addClass('self');
    row.find('.message-body').attr('data-message-id', item.id).text(item.text);
    $('.message-list').append(row);
  }
  const cache = new Map();
  function wrap(node) {
    if (!node) return null;
    if (cache.has(node)) return cache.get(node);
    const value = {getAttribute:key=>$(node).attr(key)??null,
      querySelector:selector=>wrap($(node).find(selector).get(0)),
      querySelectorAll:selector=>$(node).find(selector).toArray().map(wrap),
      classList:{contains:key=>$(node).hasClass(key)},getBoundingClientRect:()=>({width:600,height:100})};
    Object.defineProperties(value,{textContent:{get:()=>$(node).text()},
      innerText:{get:()=>$(node).text()},parentElement:{get:()=>node.parent?.type==='tag'?wrap(node.parent):null},
      hidden:{get:()=>false},inert:{get:()=>false}});
    cache.set(node,value); return value;
  }
  return {document:{querySelectorAll:selector=>$(selector).toArray().map(wrap),
    querySelector:selector=>wrap($(selector).get(0))},
    getComputedStyle:()=>({display:'flex',visibility:'visible',opacity:'1'}),TextEncoder};
}
function hashes(directory) {
  return Object.fromEntries(fs.readdirSync(directory).sort().map(name =>
    [name,crypto.createHash('sha256').update(fs.readFileSync(path.join(directory,name))).digest('hex')]));
}
function fixture(t, options={}) {
  const directory = fs.mkdtempSync(path.join(os.tmpdir(),'original-target-probe-test-'));
  t.after(()=>fs.rmSync(directory,{recursive:true,force:true}));
  const bindingsDirectory = path.join(directory,'bindings');
  const bindings = createOriginalDotBindings({directory:bindingsDirectory});
  const body = {owner:OWNER,account_id:ACCOUNT,job_id:'aibit_'+'a'.repeat(48),message,
    governance_scope:{version:1,invocation_id:'agov-fixture',work_packet_digest:packet.content_sha256},
    diagnostic_mode:'restored_target'};
  const identity = captureIdentity(body,validateDotHandoff(body,URL,OWNER));
  bindings.begin(identity,ORIGINAL);
  if (!options.noRoot) bindings.rememberRoot(identity,{target_id:ORIGINAL,root_id:ROOT,conversation_url:URL});
  if (options.canonical) bindings.rememberRoot(identity,{target_id:ORIGINAL,root_id:ROOT,
    conversation_url:'https://chatgpt.com/c/synthetic-original'});
  const before = hashes(bindingsDirectory);
  const counters = {inventory:0,account:0,snapshot:0,scope:0,ordinary:0,capture:0};
  const sessions = [];
  class Session {
    constructor(value) {this.target=value;this.closeCalls=0;this.closed=false;this.reads=0;this.locations=0;sessions.push(this);}
    async open() {if (options.open) await options.open(this);return this;}
    async location() {this.locations++;return options.location ? options.location(this) : this.target.url;}
    async evaluate(expression) {
      counters.snapshot++;this.reads++;
      if (options.evaluateError) throw Error(PRIVATE);
      const rows = options.rows ? options.rows(this) : [root(),{id:'reply',text:PRIVATE}];
      return vm.runInNewContext(expression,dom(rows),{timeout:1000});
    }
    close() {this.closeCalls++;this.closed=true;}
  }
  const listTargets = async () => {
    counters.inventory++;
    if (options.inventoryError) throw Error(PRIVATE);
    return options.targets ? options.targets(counters.inventory) : [target()];
  };
  const accountFingerprint = async session => {
    counters.account++;
    return options.account ? options.account(session,counters.account) : ACCOUNT;
  };
  const revalidateScope = async () => {
    counters.scope++;
    if (options.finalScopeError) throw Error(PRIVATE);
    return options.finalScope === undefined ? true : options.finalScope;
  };
  const probe = createRestoredTargetProbe({listTargets,Session,accountFingerprint,bindings,revalidateScope,
    config:{url:URL,owner:OWNER},...(options.now?{now:options.now}:{})});
  const run = (input=body, proof={originalIdentity:identity,scopeVerified:true}) => probe(input,proof);
  return {directory,bindingsDirectory,bindings,body,identity,before,counters,sessions,
    Session,listTargets,accountFingerprint,revalidateScope,run,
    unchanged:()=>assert.deepEqual(hashes(bindingsDirectory),before)};
}
function code(value, expected, complete=false) {
  assert.equal(value.target_probe.code,expected);
  assert.equal(value.target_probe.complete,complete);
  assert.equal(value.read_only,true);
  assert.equal(value.dot_handoff,false);
  assert.equal(value.response,'');
  assert.equal(value.root_message_id,null);
  assert.equal(value.reply_message_id,null);
  assert.equal(value.safe_to_replay,false);
}

const relayText=fs.readFileSync(path.join(source,'aibit_relay.cjs'),'utf8');
const ast=acorn.parse(relayText,{ecmaVersion:'latest'});
function functionSource(name){
  const entry=ast.body.find(node=>node.type==='FunctionDeclaration'&&node.id.name===name);
  assert.ok(entry,'Actual relay function '+name);return relayText.slice(entry.start,entry.end);
}
function actualRelay(f, options={}){
  const context={URL:global.URL,Buffer,crypto,console,setTimeout,clearTimeout,
    process:{env:{ABIT_CHATGPT_DOT_URL:URL,ABIT_CHATGPT_DOT_OWNER_ID:OWNER}},
    validateDotHandoff,captureIdentity,originalDotBindings:f.bindings,
    governedGateway:{observe:async body=>{
      f.counters.scope++;
      if(options.proofError)throw Object.assign(Error('Synthetic authority denial'),{statusCode:403});
      return {payload:body,receipt:{accepted:options.accepted!==false}};
    }},
    cdpTargets:f.listTargets,CdpSession:f.Session,accountFingerprint:f.accountFingerprint,
    aibitAccountFingerprint:f.accountFingerprint,chatThreadTabs:{},
    originalDotCaptures:{settle:()=>{f.counters.capture++;},reject:()=>{f.counters.capture++;}},
    require:name=>{
      if(name==='./original_dot_target_probe.cjs')return {createRestoredTargetProbe};
      if(name==='./aibit_dot_observation.cjs')return {createDotObserver:()=>async()=>{
        f.counters.ordinary++;return {ok:true,protocol:'ordinary-observation',response:'unchanged'};
      }};
      throw Error('Unexpected source dependency '+name);
    }};
  const observe=vm.runInNewContext(functionSource('observeOriginalDot')+'\nobserveOriginalDot',context);
  const routeContext={...context,observeOriginalDot:observe,PREFIX:'/chatgpt-relay',erpRelay:null,
    RELAY_TOKEN:'synthetic-primary',RELAY_TOKENS:['synthetic-primary','synthetic-secondary'],
    readJson:async req=>req.body,json:(res,status,body)=>{res.status=status;res.body=body;}};
  const authSources=['requestPath','tokenMatches','authorized','primaryTokenAuthorized'].map(functionSource).join('\n');
  const server=ast.body.flatMap(node=>node.type==='VariableDeclaration'?node.declarations:[])
    .find(node=>node.id.name==='server');
  const callback=server.init.arguments[0];
  const route=vm.runInNewContext(authSources+'\n('+relayText.slice(callback.start,callback.end)+')',routeContext);
  return {observe,route};
}
